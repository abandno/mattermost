### 1. **WebSocket连接建立**

````javascript path=webapp/channels/src/actions/websocket_actions.jsx mode=EXCERPT
export function initialize() {
    if (!window.WebSocket) {
        console.log('Browser does not support WebSocket');
        return;
    }

    const config = getConfig(getState());
    let connUrl = '';
    if (config.WebsocketURL) {
        connUrl = config.WebsocketURL;
    } else {
        connUrl = new URL(getSiteURL());
        // replace the protocol with a websocket one
        if (connUrl.protocol === 'https:') {
            connUrl.protocol = 'wss:';
        } else {
            connUrl.protocol = 'ws:';
        }
    }
    
    connUrl += Client4.getUrlVersion() + '/websocket';
    
    WebSocketClient.addMessageListener(handleEvent);
    WebSocketClient.initialize(connUrl, undefined, true);
}
````

每个客户端在启动时都会建立WebSocket连接到服务器，用于实时双向通信。

func (h *Hub) Broadcast(message *model.WebSocketEvent) {

        if metrics := h.platform.metricsIFace; metrics != nil {

func (ps *PlatformService) Publish(message *model.WebSocketEvent) {
    ps.PublishSkipClusterSend(message)
    
    if ps.clusterIFace != nil {
        data, err := message.ToJSON()
        if err != nil {
            mlog.Warn("Failed to encode message to JSON", mlog.Err(err))
        }
        cm := &model.ClusterMessage{
            Event:    model.ClusterEventPublish,
            SendType: model.ClusterSendBestEffort,
            Data:     data,
        }
        
        if message.EventType() == model.WebsocketEventPosted ||
            message.EventType() == model.WebsocketEventPostEdited {
            cm.SendType = model.ClusterSendReliable
        }
        
        ps.clusterIFace.SendClusterMessage(cm)
    }
}

### 5. **防抖处理和批量更新**

为了优化性能，Mattermost使用防抖机制处理大量消息：

````javascript path=webapp/channels/src/actions/websocket_actions.jsx mode=EXCERPT
function debouncePostEvent(wait) {
    let timeout;
    let queue = [];
    let count = 0;
    
    const triggered = () => {
        timeout = null;
        
        if (queue.length > 0) {
            dispatch(handleNewPostEvents(queue));
        }
        
        queue = [];
        count = 0;
    };
    
    return function fx(msg) {
        if (timeout && count > 4) {
            // Queue up multiple events
            if (queue.push(msg) > 200) {
                queue = [];
                console.log('channel broken because of too many incoming messages');
            }
            clearTimeout(timeout);
            timeout = setTimeout(triggered, wait);
        } else {
            // Apply immediately for first few events
            count += 1;
            dispatch(handleNewPostEvent(msg));
            clearTimeout(timeout);
            timeout = setTimeout(triggered, wait);
        }
    };
}

const handleNewPostEventDebounced = debouncePostEvent(100);
````

### 6. **Redux Store更新**

接收到的消息会更新Redux store，触发UI重新渲染：

````typescript path=webapp/channels/src/packages/mattermost-redux/src/reducers/entities/posts.ts mode=EXCERPT
export function handlePosts(state: IDMappedObjects<Post> = {}, action: MMReduxAction) {
    switch (action.type) {
    case PostTypes.RECEIVED_POST:
    case PostTypes.RECEIVED_NEW_POST: {
        return handlePostReceived({...state}, action.data);
    }
    
    case PostTypes.RECEIVED_POSTS: {
        const posts = Object.values(action.data.posts) as Post[];
        
        if (posts.length === 0) {
            return state;
        }
        
        const nextState = {...state};
        
        for (const post of posts) {
            handlePostReceived(nextState, post);
        }
        
        return nextState;
    }
    }
}
````

## 总结

Mattermost的实时消息传递机制包括：

1. **WebSocket长连接**：每个客户端与服务器建立持久的WebSocket连接
2. **服务端广播**：当有新post时，服务器立即通过WebSocket广播给所有相关用户
3. **集群同步**：在多服务器环境中，消息会可靠地同步到所有节点
4. **客户端处理**：客户端接收消息后，通过防抖机制优化性能，批量处理消息
5. **状态更新**：消息更新Redux store，触发React组件重新渲染，用户立即看到新消息

这种架构确保了消息的实时性和可靠性，让用户能够立即看到其他人发送的消息。
